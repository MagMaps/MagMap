// ------------ FUNCTION: Calculate Monthly NDVI ------------ //
function getMonthlyNDVI(aoi, month, year, aoiName) {
  var start = ee.Date.fromYMD(year, month, 1);
  var end = start.advance(1, 'month');
  
  var s2 = ee.ImageCollection('COPERNICUS/S2_SR_HARMONIZED')
              .filterBounds(aoi)
              .filterDate(start, end)
              .filter(ee.Filter.lt('CLOUDY_PIXEL_PERCENTAGE', 30));
  
  var imageCount = s2.size();
  print(aoiName, 'Month', month, '- Images found:', imageCount);
  
  // Only proceed if we have images
  var hasImages = imageCount.gt(0);
  
  return ee.Algorithms.If(
    hasImages,
    function() {
      var composite = s2.median();
      var ndvi = composite.normalizedDifference(['B8', 'B4']).rename('NDVI');
      
      // Calculate statistics with proper parameters
      var stats = ndvi.reduceRegion({
        reducer: ee.Reducer.mean().combine({
          reducer2: ee.Reducer.count(),
          sharedInputs: true
        }),
        geometry: aoi,
        scale: 10,
        maxPixels: 1e9,
        bestEffort: true,
        tileScale: 4
      });
      
      return ee.Feature(null, {
        class: aoiName,
        year: year,
        month: month,
        date: ee.String(ee.Number(year).format()).cat('-').cat(ee.Number(month).format('%02d')),
        ndvi_mean: stats.get('NDVI_mean'),
        pixel_count: stats.get('NDVI_count'),
        image_count: imageCount
      });
    },
    function() {
      return ee.Feature(null, {
        class: aoiName,
        year: year,
        month: month,
        date: ee.String(ee.Number(year).format()).cat('-').cat(ee.Number(month).format('%02d')),
        ndvi_mean: null,
        pixel_count: 0,
        image_count: 0
      });
    }
  )();
}

// ------------ FUNCTION: Calculate Band Statistics ------------ //
function getBandStats(aoi, month, year, aoiName) {
  var start = ee.Date.fromYMD(year, month, 1);
  var end = start.advance(1, 'month');
  
  var s2 = ee.ImageCollection('COPERNICUS/S2_SR_HARMONIZED')
              .filterBounds(aoi)
              .filterDate(start, end)
              .filter(ee.Filter.lt('CLOUDY_PIXEL_PERCENTAGE', 30));
  
  var imageCount = s2.size();
  
  return ee.Algorithms.If(
    imageCount.gt(0),
    function() {
      var composite = s2.median();
      
      var stats = composite.reduceRegion({
        reducer: ee.Reducer.mean(),
        geometry: aoi,
        scale: 20, // Use 20m for faster processing
        maxPixels: 1e9,
        bestEffort: true,
        tileScale: 4
      });
      
      return ee.Feature(null, {
        class: aoiName,
        year: year,
        month: month,
        image_count: imageCount,
        B1: stats.get('B1'),
        B2: stats.get('B2'),
        B3: stats.get('B3'),
        B4: stats.get('B4'),
        B5: stats.get('B5'),
        B6: stats.get('B6'),
        B7: stats.get('B7'),
        B8: stats.get('B8'),
        B8A: stats.get('B8A'),
        B9: stats.get('B9'),
        B11: stats.get('B11'),
        B12: stats.get('B12')
      });
    },
    function() {
      return ee.Feature(null, {
        class: aoiName,
        year: year,
        month: month,
        image_count: 0,
        B1: null, B2: null, B3: null, B4: null, B5: null, B6: null,
        B7: null, B8: null, B8A: null, B9: null, B11: null, B12: null
      });
    }
  )();
}
