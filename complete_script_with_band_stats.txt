// ------------ FUNCTION: Monthly NDVI Composite ------------ //
function getMonthlyNDVI(aoi, month, year) {
  var start = ee.Date.fromYMD(year, month, 1);
  var end   = start.advance(1, 'month');
  
  var s2 = ee.ImageCollection('COPERNICUS/S2_SR_HARMONIZED')
              .filterBounds(aoi)
              .filterDate(start, end)
              .filter(ee.Filter.lt('CLOUDY_PIXEL_PERCENTAGE', 50));
  
  var imageCount = s2.size();
  
  // If no images, try without cloud filter
  s2 = ee.Algorithms.If(
    imageCount.eq(0),
    ee.ImageCollection('COPERNICUS/S2_SR_HARMONIZED')
      .filterBounds(aoi)
      .filterDate(start, end),
    s2
  );
  
  imageCount = ee.ImageCollection(s2).size();
  
  // Create a dummy NDVI image if no data exists
  var ndvi = ee.Algorithms.If(
    imageCount.gt(0),
    ee.ImageCollection(s2).median().clip(aoi).normalizedDifference(['B8','B4']).rename('NDVI'),
    ee.Image.constant(-9999).rename('NDVI')
  );
  
  return ee.Image(ndvi).set({
    'month': month, 
    'year': year,
    'system:time_start': start.millis(),
    'image_count': imageCount
  });
}

// ------------ NEW FUNCTION: Monthly All-Band Composite ------------ //
function getMonthlyAllBands(aoi, month, year) {
  var start = ee.Date.fromYMD(year, month, 1);
  var end   = start.advance(1, 'month');
  
  var s2 = ee.ImageCollection('COPERNICUS/S2_SR_HARMONIZED')
              .filterBounds(aoi)
              .filterDate(start, end)
              .filter(ee.Filter.lt('CLOUDY_PIXEL_PERCENTAGE', 50));
  
  var imageCount = s2.size();
  
  // If no images, try without cloud filter
  s2 = ee.Algorithms.If(
    imageCount.eq(0),
    ee.ImageCollection('COPERNICUS/S2_SR_HARMONIZED')
      .filterBounds(aoi)
      .filterDate(start, end),
    s2
  );
  
  imageCount = ee.ImageCollection(s2).size();
  
  // Create composite with all bands or dummy image
  var composite = ee.Algorithms.If(
    imageCount.gt(0),
    ee.ImageCollection(s2).median().clip(aoi),
    ee.Image.constant(-9999).rename(['B1', 'B2', 'B3', 'B4', 'B5', 'B6', 'B7', 'B8', 'B8A', 'B9', 'B11', 'B12'])
  );
  
  return ee.Image(composite).set({
    'month': month, 
    'year': year,
    'system:time_start': start.millis(),
    'image_count': imageCount
  });
}

// ------------ EXPORT FUNCTION: NDVI ------------ //
function exportNDVIMonthlyStats(series, aoiGeom, className) {
  var features = series.map(function(img) {
    var month = img.get('month');
    var year = img.get('year');
    var imageCount = img.get('image_count');
    
    var ndviMean = img.reduceRegion({
      reducer: ee.Reducer.mean(),
      geometry: aoiGeom,
      scale: 10,
      maxPixels: 1e9,
      bestEffort: true
    }).get('NDVI');
    
    return ee.Feature(null, {
      class: className,
      year: year,
      month: month,
      date: ee.String(ee.Number(year).format()).cat('-').cat(ee.Number(month).format('%02d')),
      ndvi_mean: ndviMean,
      image_count: imageCount
    });
  });
  
  Export.table.toDrive({
    collection: ee.FeatureCollection(features),
    description: className + '_NDVI_Stats_2023',
    folder: 'EE_Seasonal_Analysis',
    fileFormat: 'CSV'
  });
  
  print('NDVI export task created for:', className);
}

// ------------ NEW EXPORT FUNCTION: ALL BANDS MONTHLY ------------ //
function exportAllBandsMonthlyStats(series, aoiGeom, className) {
  var features = series.map(function(img) {
    var month = img.get('month');
    var year = img.get('year');
    var imageCount = img.get('image_count');
    
    var stats = img.reduceRegion({
      reducer: ee.Reducer.mean().combine({
        reducer2: ee.Reducer.stdDev(),
        sharedInputs: true
      }),
      geometry: aoiGeom,
      scale: 10,
      maxPixels: 1e9,
      bestEffort: true
    });
    
    return ee.Feature(null, {
      class: className,
      year: year,
      month: month,
      date: ee.String(ee.Number(year).format()).cat('-').cat(ee.Number(month).format('%02d')),
      image_count: imageCount,
      // All band means
      B1_mean: stats.get('B1_mean'),
      B2_mean: stats.get('B2_mean'),
      B3_mean: stats.get('B3_mean'),
      B4_mean: stats.get('B4_mean'),
      B5_mean: stats.get('B5_mean'),
      B6_mean: stats.get('B6_mean'),
      B7_mean: stats.get('B7_mean'),
      B8_mean: stats.get('B8_mean'),
      B8A_mean: stats.get('B8A_mean'),
      B9_mean: stats.get('B9_mean'),
      B11_mean: stats.get('B11_mean'),
      B12_mean: stats.get('B12_mean'),
      // All band standard deviations
      B1_stdDev: stats.get('B1_stdDev'),
      B2_stdDev: stats.get('B2_stdDev'),
      B3_stdDev: stats.get('B3_stdDev'),
      B4_stdDev: stats.get('B4_stdDev'),
      B5_stdDev: stats.get('B5_stdDev'),
      B6_stdDev: stats.get('B6_stdDev'),
      B7_stdDev: stats.get('B7_stdDev'),
      B8_stdDev: stats.get('B8_stdDev'),
      B8A_stdDev: stats.get('B8A_stdDev'),
      B9_stdDev: stats.get('B9_stdDev'),
      B11_stdDev: stats.get('B11_stdDev'),
      B12_stdDev: stats.get('B12_stdDev')
    });
  });
  
  Export.table.toDrive({
    collection: ee.FeatureCollection(features),
    description: className + '_AllBands_Monthly_2023',
    folder: 'EE_Seasonal_Analysis',
    fileFormat: 'CSV'
  });
  
  print('All-bands monthly export task created for:', className);
}
